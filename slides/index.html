const parser = createJsonStreamParser({
  onField: (key, value) => store.field(key, value),        // "region" → volar el mapa
  onPartialString: (key, text) => store.partial(key, text), // título letra a letra

});

// Streaming · 2

Los tokens llegan cuando quieren. Tú pintas por frame.

genui/doc-store.ts
private mark(key) {
  this.changed.add(key);
  if (this.scheduled) return;
  this.scheduled = true;
  requestAnimationFrame(() => this.flush());
  // rAF se pausa en pestañas ocultas:
  setTimeout(() => this.flush(), 60);
}

genui/a11y.ts
export function createAnnouncer(region, minGapMs = 2500) {
  region.setAttribute('aria-live', 'polite');
  return {
    say(message, force = false) {
      // pocos, cortos y en momentos clave
      if (force || elapsed() >= minGapMs) write(message);
      else schedule(message);
    },
  };
}

// Para llevar

10 reglas para una IA rápida y una UI fluida

  1. Datos, no HTML: catálogo de componentes + JSON Schema.
  2. Reserva el espacio antes del contenido (slots, aspect-ratio, line-clamp).
  3. Parser incremental: cada carácter una vez.
  4. Un render por frame: rAF + respaldo; cede el hilo con scheduler.yield().
  5. Claves estables por id; mueve con View Transitions.
  6. Nunca pantalla en blanco: lo anterior se queda hasta su reemplazo.
  7. La IA decide, el código calcula (y verifica).
  8. Anuncios con intención: aria-busy y mensajes espaciados.
  9. Local primero: Gemini Nano + nube de respaldo + modo simulado.
  10. Mide con DevTools: pistas propias, AI assistance y agentes.
